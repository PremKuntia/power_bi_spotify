-- some important syntax -- 
-- Create a Date Table to get the real time and unique dates for each start to end date (Very Important)

Date Table = Calender(min(main column), max(main column))

-- get the min and max values for a line chart

MinMax Albumns Line Chart = 
VAR _maxValue = MAXX(ALLSELECTED('Date Table'[Year]) , CALCULATE(DISTINCTCOUNT ('public spotify_bi'[album_name])))

VAR _minValue = MINX(ALLSELECTED('Date Table'[Year]), CALCULATE(DISTINCTCOUNT ('public spotify_bi'[album_name])))

VAR _currentValue =  CALCULATE(DISTINCTCOUNT ('public spotify_bi'[album_name]))
RETURN 
IF(_currentValue = _maxValue || _currentValue = _minValue, _currentValue, BLANK())

-- current Year record 
CurrentYearAlbums = 
VAR _LatestYear = MAX('Date Table'[Year])
RETURN 
CALCULATE(DISTINCTCOUNT(spotify_history[album_name]), 'Date Table'[Year] = _LatestYear)

-- Previous Year record
PreviousYearAlbums = 
VAR _LatestYear = MAX('Date Table'[Year])
VAR _PreviousYear = _LatestYear - 1
RETURN 
CALCULATE(DISTINCTCOUNT(spotify_history[album_name]), 'Date Table'[Year] = _PreviousYear)

-- previous year and yoy album KPIs 
PY and YoY Albums KPI = 

VAR _latest = [LatestYearAlbums]
VAR _previous = [PreviousYearAlbums]
VAR _YoY = If(not(isblank(_previous)), divide(_latest - _previous, _previous, 0), BlANK())
RETURN 
IF(NOT(ISBLANK(_previous)),
"vs PY: " & FORMAT(_previous, "#,##0") & 
"(" & FORMAT(_YoY, "0.00%") & ")",
"No Data")

-- comparing with latest and previous year
CurrentYearAlbumNew = 
VAR _CurrentYear = 
CALCULATE(MAX('DATE Table'[Year]),
'Date Table', -- ensures the date table is used properly
TREATAS(Values(spotify_history[Year pf Track Date]), 'Date Table'[Year]), -- Ensures filtering works properly
VALUES(spotify[platform]) --Ensures we respect the selected platform
)
RETURN
CALCULATE(DISTINCTCOUNT(spotify_history[album_name]),
'Date Table'[Year] = _CurrentYear)
